decl
    int answer;
    int factorial(int n);
enddecl

int factorial(int n){
decl
    int temp;
enddecl
begin
    if(n == 0) then
        temp = 1;
    else
        temp = n * factorial(n - 1);
    endif;
return temp;
end
}

int main(){
decl
enddecl
begin
    answer = factorial(5);
    write(answer);
return 0;
end
}
